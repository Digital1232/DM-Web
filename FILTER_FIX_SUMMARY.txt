FIXED CODE (New):
────────────────────────────────────────────────────────────────────────────
function renderEmployeeClientTimingReport() {
    const filters = getEcttFilters();
    
    // SMART DEFAULT LOGIC
    let fromTs, toTs;
    
    if (reportDateFrom && reportDateTo) {
        // Use global filters if set
        fromTs = new Date(reportDateFrom).getTime();
        toTs = new Date(reportDateTo).getTime() + 86400000;
    } else {
        // Use default: last 30 days
        const now = Date.now();
        fromTs = now - (30 * 86400000);
        toTs = now;
    }
    
    // Always continues - no blocking
    // Render data...
}
